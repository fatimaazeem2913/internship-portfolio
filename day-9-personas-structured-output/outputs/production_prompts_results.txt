import heapq
from typing import Generator, List, Union


def running_median(
    stream: List[Union[int, float]]
) -> Generator[Union[int, float], None, None]:
    """Calculates the running median of a stream of numbers.

    Parameters:
    - stream (List[Union[int, float]]): An iterable or list of numbers representing the stream.

    Yields:
    - Union[int, float]: The median of the numbers seen so far.

    Edge Cases:
    - If the stream is empty (contains no elements), calling `next()` on the generator
      or iterating over it will yield nothing. If an immediate result is needed for an
      empty stream, calling `next(gen, None)` handles this gracefully.

    Complexity:
    - Time: O(log N) per number, where N is the number of elements processed so far.
    - Space: O(N) to store the elements in two heaps.
    """
    if not stream:
        return

    # max_heap stores the smaller half of the numbers (inverted to use Python's min-heap)
    max_heap: List[Union[int, float]] = []
    # min_heap stores the larger half of the numbers
    min_heap: List[Union[int, float]] = []

    for num in stream:
        # Step 1: Add to heaps
        if not max_heap or num <= -max_heap[0]:
            heapq.heappush(max_heap, -num)
        else:
            heapq.heappush(min_heap, num)

        # Step 2: Balance the heaps
        # max_heap can have at most one more element than min_heap
        if len(max_heap) > len(min_heap) + 1:
            heapq.heappush(min_heap, -heapq.heappop(max_heap))
        elif len(min_heap) > len(max_heap):
            heapq.heappush(max_heap, -heapq.heappop(min_heap))

        # Step 3: Compute median
        if len(max_heap) == len(min_heap):
            yield (-max_heap[0] + min_heap[0]) / 2.0
        else:
            yield float(-max_heap[0])
